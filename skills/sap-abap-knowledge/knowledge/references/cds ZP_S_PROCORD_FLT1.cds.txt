define view ZP_S_PROCORD_FLT1
  as select from ZP_S_DCB_PROCORD_OPER as procord
    inner join   Zp_po_rt          as rt on  procord.werks = rt.werks
                                                 and procord.auart = rt.auart
                                                 and procord.phseq = rt.phseq

{

  key        OrderID            as aufnr,
  key        OrderOperation,
  key        rt.werks,
  key        rt.auart,


  key        rt.phseq,
             rt.business_system as mes,
             plnbez,

             -- OrderCategory
